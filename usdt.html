import qrcode from qrcode.image.styledpil import StyledPilImage from qrcode.image.styles.moduledrawers import RoundedModuleDrawer from PIL import Image, ImageDraw # 🔗 Link do seu site — atualize depois que o Pages estiver ativo url = "https://rvlogistca.github.io/ustd/" # Cria QR Code qr = qrcode.QRCode( version=1, error_correction=qrcode.constants.ERROR_CORRECT_H, box_size=12, border=4, ) qr.add_data(url) qr.make(fit=True) # Base com cantos arredondados img = qr.make_image( image_factory=StyledPilImage, module_drawer=RoundedModuleDrawer() ).convert("RGBA") # Logo USTD no centro def create_ustd_logo(size=140): logo = Image.new("RGBA", (size, size), (0, 0, 0, 0)) draw = ImageDraw.Draw(logo) c = (size//2, size//2) r = size//2 - 5 # Círculo verde draw.ellipse([c[0]-r, c[1]-r, c[0]+r, c[1]+r], fill="#0a6648") # Borda dourada draw.ellipse([c[0]-r+4, c[1]-r+4, c[0]+r-4, c[1]+r-4], outline="#d4af37", width=5) # Letra T draw.rectangle([c[0]-30, c[1]-35, c[0]+30, c[1]-17], fill="#d4af37") draw.rectangle([c[0]-8, c[1]-35, c[0]+8, c[1]+35], fill="#d4af37") # Anel draw.ellipse([c[0]-45, c[1]-18, c[0]+45, c[1]+18], outline="#d4af37", width=4) return logo ustd_logo = create_ustd_logo(140) pos = ((img.size[0]-140)//2, (img.size[1]-140)//2) # Fundo branco atrás da logo bg = Image.new("RGBA", img.size, (0,0,0,0)) bg_draw = ImageDraw.Draw(bg) center = (img.size[0]//2, img.size[1]//2) bg_draw.ellipse([center[0]-85, center[1]-85, center[0]+85, center[1]+85], fill="white") # Monta tudo img = Image.alpha_composite(img, bg) img.paste(ustd_logo, pos, ustd_logo) # Salva img.convert("RGB").save("ustd_qrcode.png") print("✅ Salvo: ustd_qrcode.png | Link:", url)

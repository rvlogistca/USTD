usdt_ethers_v6.js 
import { ethers } from "ethers";

// 1. Configurações do USDT no Ethereum (ERC-20)
const USDT_ADDRESS = "0xdAC17F958D2ee523a2206206994597C13D831ec7";
const ERC20_ABI = [
  "function balanceOf(address account) external view returns (uint256)",
  "function decimals() external view returns (uint8)",
  "function transfer(address recipient, uint256 amount) external returns (bool)"
];

// 2. Função para consultar saldo em USDT
async function getUsdtBalance(walletAddress) {
  // Provedor de leitura de nós (RPC público)
  const provider = new ethers.JsonRpcProvider("https://eth.llamarpc.com");
  
  // Instanciação do contrato
  const contract = new ethers.Contract(USDT_ADDRESS, ERC20_ABI, provider);
  
  // Leitura do saldo em unidades brutas (6 casas decimais)
  const rawBalance = await contract.balanceOf(walletAddress);
  const decimals = await contract.decimals();
  
  // Conversão para formato legível (ex: 100.50)
  const formattedBalance = ethers.formatUnits(rawBalance, decimals);
  console.log(`Saldo de USDT: ${formattedBalance}`);
  return formattedBalance;
}

usdt_tracker.py 
from web3 import Web3

# Conexão com o nó Ethereum
rpc_url = "https://eth.llamarpc.com"
w3 = Web3(Web3.HTTPProvider(rpc_url))

# Endereço e ABI mínima do USDT
usdt_address = Web3.to_checksum_address("0xdAC17F958D2ee523a2206206994597C13D831ec7")
min_abi = [
    {
        "constant": True,
        "inputs": [{"name": "_owner", "type": "address"}],
        "name": "balanceOf",
        "outputs": [{"name": "balance", "type": "uint256"}],
        "type": "function"
    },
    {
        "constant": True,
        "inputs": [],
        "name": "decimals",
        "outputs": [{"name": "", "type": "uint8"}],
        "type": "function"
    }
]

# Inicializa o contrato
contract = w3.eth.contract(address=usdt_address, abi=min_abi)

def consultar_saldo_usdt(endereco_carteira):
    wallet = Web3.to_checksum_address(endereco_carteira)
    raw_balance = contract.functions.balanceOf(wallet).call()
    decimals = contract.functions.decimals().call()
    
    # USDT utiliza 6 casas decimais
    saldo_formatado = raw_balance / (10 ** decimals)
    return saldo_formatado

USDTPaymentGateway.sol 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

interface IERC20 {
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
}

/**
 * @title Processador de Pagamentos em USDT
 * @dev Permite receber USDT de clientes via transferência aprovada
 */
contract USDTPaymentGateway {
    address public owner;
    IERC20 public immutable usdtToken;

    event PaymentReceived(address indexed customer, uint256 amount, string orderId);

    constructor(address _usdtContractAddress) {
        owner = msg.sender;
        usdtToken = IERC20(_usdtContractAddress);
    }

    /**
     * @notice Processa um pagamento do cliente
     * @param amount Quantidade em unidades brutas (ex: 10 USDT = 10000000)
     * @param orderId Identificador único do pedido
     */
    function processPayment(uint256 amount, string calldata orderId) external {
        require(amount > 0, "Valor deve ser maior que zero");
        
        // Transfere os tokens da carteira do cliente para o contrato
        bool success = usdtToken.transferFrom(msg.sender, address(this), amount);
        require(success, "Falha na transferencia de USDT");

        emit PaymentReceived(msg.sender, amount, orderId);
    }
}
tronweb_usdt.js 
// TRON TRC-20 USDT Integration (TronWeb)
const USDT_TRC20_CONTRACT = "TR7NHqjeKQxGTCi8q8ZY4pL8otSzgjLj6t";

async function sendTRC20USDT(recipientAddress, amountUsdt) {
  try {
    if (!window.tronWeb || !window.tronWeb.defaultAddress.base58) {
      throw new Error("TronLink / TronWeb nao detectado.");
    }

    // Instancia o contrato TRC-20
    let contract = await window.tronWeb.contract().at(USDT_TRC20_CONTRACT);
    
    // Converte para 6 casas decimais (SUN units)
    let amountSun = TronWeb.toBigNumber(amountUsdt).multipliedBy(1e6).toString();
    
    // Envia a transacao
    let result = await contract.transfer(recipientAddress, amountSun).send();
    console.log("HASH da Transacao TRON:", result);
    return result;
  } catch (error) {
    console.error("Erro ao enviar TRC20 USDT:", error);
  }
}
